#### 2. 状态管理规范

```typescript
// Store设计规范
class TodoStore {
  // 1. 状态定义 - 使用私有属性
  private _todos: TodoItem[] = []
  private _loading: boolean = false
  private _error: string | null = null
  
  // 2. 状态访问器 - 提供只读访问
  get todos(): readonly TodoItem[] {
    return this._todos
  }
  
  get loading(): boolean {
    return this._loading
  }
  
  get error(): string | null {
    return this._error
  }
  
  // 3. 计算属性
  get completedTodos(): readonly TodoItem[] {
    return this._todos.filter(todo => todo.status === TodoStatus.COMPLETED)
  }
  
  get todoStats(): TodoStats {
    return {
      total: this._todos.length,
      completed: this.completedTodos.length,
      pending: this._todos.length - this.completedTodos.length
    }
  }
  
  // 4. 同步方法 - 纯函数，无副作用
  addTodo(todo: TodoItem): void {
    this._todos = [...this._todos, todo]
    this.notifyUpdate()
  }
  
  updateTodo(id: string, updates: Partial<TodoItem>): void {
    this._todos = this._todos.map(todo => 
      todo.id === id ? { ...todo, ...updates } : todo
    )
    this.notifyUpdate()
  }
  
  removeTodo(id: string): void {
    this._todos = this._todos.filter(todo => todo.id !== id)
    this.notifyUpdate()
  }
  
  // 5. 异步方法 - 处理副作用
  async loadTodos(): Promise<void> {
    try {
      this._loading = true
      this._error = null
      
      const todos = await this.todoRepository.findAll()
      this._todos = todos
      
    } catch (error) {
      this._error = error.message
      console.error('加载待办失败:', error)
    } finally {
      this._loading = false
      this.notifyUpdate()
    }
  }
  
  async saveTodo(todo: TodoItem): Promise<boolean> {
    try {
      await this.todoRepository.save(todo)
      this.addTodo(todo)
      return true
    } catch (error) {
      this._error = error.message
      console.error('保存待办失败:', error)
      return false
    }
  }
  
  // 6. 私有辅助方法
  private notifyUpdate(): void {
    // 通知组件更新
    AppStorage.Set('todoStore', this)
  }
}
```

```typescript
@Component
struct ResponsiveComponent {
  @StorageLink('windowWidth') windowWidth: number = 360
  
  // 断点定义
  private readonly BREAKPOINTS = {
    xs: 0,      // 超小屏
    sm: 600,    // 小屏
    md: 840,    // 中屏  
    lg: 1024,   // 大屏
    xl: 1440    // 超大屏
  }
  
  // 计算当前屏幕类型
  get screenType(): 'xs' | 'sm' | 'md' | 'lg' | 'xl' {
    if (this.windowWidth >= this.BREAKPOINTS.xl) return 'xl'
    if (this.windowWidth >= this.BREAKPOINTS.lg) return 'lg'
    if (this.windowWidth >= this.BREAKPOINTS.md) return 'md'
    if (this.windowWidth >= this.BREAKPOINTS.sm) return 'sm'
    return 'xs'
  }
  
  // 响应式列数
  get columnCount(): number {
    switch (this.screenType) {
      case 'xs': return 1
      case 'sm': return 2
      case 'md': return 3
      case 'lg': return 4
      case 'xl': return 5
      default: return 1
    }
  }
  
  // 响应式内边距
  get padding(): number {
    return this.screenType === 'xs' ? 12 : 16
  }
  
  build() {
    Grid() {
      ForEach(this.items, (item: any) => {
        GridItem() {
          this.ItemCard(item)
        }
      })
    }
    .columnsTemplate(`repeat(${this.columnCount}, 1fr)`)
    .rowsGap(12)
    .columnsGap(12)
    .padding(this.padding)
  }
}
```

```typescript
@Component
struct OptimizedList {
  @State items: ListItem[] = []
  @State visibleRange: { start: number, end: number } = { start: 0, end: 20 }
  
  // 虚拟列表实现
  build() {
    List() {
      // 使用LazyForEach实现虚拟滚动
      LazyForEach(this.dataSource, (item: ListItem, index: number) => {
        ListItem() {
          if (this.isItemVisible(index)) {
            this.ItemComponent(item)
          } else {
            // 占位符
            Row()
              .height(this.getItemHeight(item))
              .width('100%')
          }
        }
      }, (item) => item.id)
    }
    .onScroll((scrollOffset: number, scrollState: ScrollState) => {
      this.updateVisibleRange(scrollOffset)
    })
    .cachedCount(5) // 缓存数量
  }
  
  @Builder ItemComponent(item: ListItem) {
    // 使用@Reusable装饰器标记可复用组件
    ReusableItemCard({ item: item })
  }
  
  private isItemVisible(index: number): boolean {
    return index >= this.visibleRange.start && index <= this.visibleRange.end
  }
  
  private updateVisibleRange(scrollOffset: number) {
    // 更新可见区域范围
    const itemHeight = 80
    const start = Math.max(0, Math.floor(scrollOffset / itemHeight) - 5)
    const end = Math.min(this.items.length - 1, start + 30)
    
    if (this.visibleRange.start !== start || this.visibleRange.end !== end) {
      this.visibleRange = { start, end }
    }
  }
}
```

```typescript
      { id: '2', title: 'Todo 2', status: TodoStatus.COMPLETED }
    ]
    
    mockRepository.findAll.mockResolvedValue(mockTodos)
    
    await todoStore.loadTodos()
    
    expect(todoStore.todos).toEqual(mockTodos)
    expect(mockRepository.findAll).toHaveBeenCalled()
  })

    expect(todoStore.loading).toBe(false)
  })
```
